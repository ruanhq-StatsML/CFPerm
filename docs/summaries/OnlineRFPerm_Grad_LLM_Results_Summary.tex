\begin{abstract}
We summarize the \textbf{latest} OnlineRFPerm (OnlinePermOOB) monitoring
results after adding an extra scalar information stream into the same
online rank/EWMA + FDR shell.
Naming: the repo implements \textbf{Grad-OnlineRFPerm}
($g_t=\|\gradU L_t\|_2$), not a separate ``OnlineFRPerm'' binary.
\textbf{LLM / TabPFN backends are designed but not yet landed}; the
landed fusion is the \emph{gradient-energy} gate.
Primary real-data timing metric is
$\Lead=t_{\mathrm{grad}}-t_{\mathrm{MSE}}$ (negative $=$ Grad earlier).
Baseline alarm rate $\AR=n_{\mathrm{alarm}}/n_{\mathrm{batch}}$ is a
\emph{reference}, \textbf{not} Type-I FAR (ongoing updates break any
stationary-null claim).
\end{abstract}

\section{How information enters OnlineRFPerm}
\label{sec:fuse}

\paragraph{Locked shell (unchanged).}
Burn-in reference $\to$ per-batch scalar $T_t$ $\to$ online $p_t$
(rank / EWMA) $\to$ online FDR $\Rightarrow$ reject $R_t$.
Parallel streams: MSE / MMD / PO OnlineRFPerm for comparison.

\paragraph{Landed fusion: Grad.}
\[
  g_t=\|\gradU L_t\|_2,\qquad
  T_t^{\mathrm{grad}}=g_t-e_{\mathrm{ref}},
\]
with $e_{\mathrm{ref}}$ updated on burn-in only (frozen reference
energy). Single FDR stream (no per-layer multiplicity).

\paragraph{Planned fusion: LLM / TabPFN (not coded).}
Same shell; pluggable backend
$\{\mathrm{RF},\,\mathrm{TabPFN},\,\mathrm{LLM\text{-}embed+head}\}$.
LLM path: frozen embedding $\to$ light head $\to$ Grad or MSE scalar
into OnlineRFPerm
(\texttt{docs/method/Sep17\_daily\_three\_items.md}:
``\emph{尚无} TabPFN / LLM backend'').
\textbf{No with-LLM vs without-LLM table exists yet.}
Numbers below are Grad vs MSE (and peers), i.e.\ the current
``extra information'' ablation.

\paragraph{Protocol (all tables unless noted).}
$b{=}8$, batch $128$, $48$ batches, $\alpha{=}0.05$,
seeds $\{0,\ldots,4\}$ (extras often $\{0,1,2\}$).

\section{Baseline alarm rate (FAR-proxy)}
\label{sec:ar}

\begin{equation}
\label{eq:AR}
  \AR
  \;=\;
  \frac{\#\{\text{reject batches}\}}{\#\{\text{post-burn observation batches}\}}.
\end{equation}
Near-stationary \emph{reference} synthetic stream (serving still
updates). \textbf{Do not} read as Type-I FAR.

\begin{table}[h]
\centering
\caption{Reference-stream baseline alarm rate (Grad vs MSE).
Grace $g$ skips the first $g$ post-burn batches in the AR window.}
\label{tab:ar}
\begin{tabular}{crrrr}
\toprule
grace $g$ & Grad $\AR$ & MSE $\AR$ & mean $n_{\mathrm{alarm}}/n_{\mathrm{batch}}$ & mean $t_{\mathrm{grad}}$ \\
\midrule
$0$ & $0.233$ & $0.225$ & $9.3/40$ & $8.0$ \\
$2$ & $0.211$ & $0.228$ & $8.0/38$ & $10.7$ \\
$4$ & $0.204$ & $0.222$ & $7.3/36$ & $15.3$ \\
\bottomrule
\end{tabular}
\end{table}

\noindent\textit{Takeaway.} Grad $\AR\approx$ MSE $\AR$
($\sim 0.20$--$0.23$). Grace mildly delays first reject
($8\to 15$). Higher AR is acceptable as a baseline under updates.

\section{Detection timing on real data}
\label{sec:delay}

\subsection{Definitions}
\begin{itemize}[leftmargin=*,itemsep=0.25em]
  \item \textbf{First-reject time} $t_{\bullet}$: first FDR reject batch
        index (post stream start; burn $b{=}8$).
  \item $\displaystyle\Lead=t_{\mathrm{grad}}-t_{\mathrm{MSE}}$.
        $\Lead<0$ $\Leftrightarrow$ Grad earlier than MSE-OnlineRFPerm.
  \item \textbf{Detection delay} after a \emph{known} shift is available
        only on \textbf{synthetic} (shift@20): all methods report
        $\mathrm{delay}=0$ (first post-shift reject at the shift batch).
        Real OpenML packs have \texttt{detection\_delay=null}; we report
        first-reject / Lead distributions instead.
\end{itemize}
