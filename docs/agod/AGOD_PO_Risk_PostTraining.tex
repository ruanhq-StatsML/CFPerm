\begin{abstract}
\textbf{Goal (only).}
Accelerate \emph{post-training} / online adaptation:
reach a target holdout quality with fewer update FLOPs / wall-clock /
optimizer steps than uniform dense fine-tuning.

\textbf{Role of metrics.}
PO-risk (and optional MMD / proto / $\Delta\PO$) are not leaderboard
scores; they are \emph{sensors} whose landing procedure is
\[
  \text{metric}_t
  \;\xrightarrow{\;\text{map}\;}
  \text{actuators}_{t+1}
  \;\xrightarrow{\;\text{train}\;}
  \text{speed KPIs}.
\]
Primary report: $\Flops$, step budget used, wall-clock to
$\Acc\ge\Acc^\star$.
Secondary (constraint): $\Acc$ / MSE must not collapse vs equal.
This note writes the landing procedure for each metric version with
that focus.
\end{abstract}

\section{Goal lock: accelerate post-training}
\label{sec:goal}

\begin{definition}[Post-training acceleration]
Fix a pretrained / distilled reference $f_{\mathrm{ref}}$.
On a stream of windows $t=1,2,\ldots$, a policy $\pi$ sets
next-window update knobs.
Acceleration means, relative to dense equal post-train $\pi_0$:
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item \textbf{Primary} --- lower update cost to hit a quality bar:
        \[
          T_{\pi}(\Acc^\star)
          \;=\;
          \min\bigl\{\,T:\ \Acc_T\ge\Acc^\star\,\bigr\}
          \quad\text{smaller is better},
        \]
        or fixed-$T$ cost $\sum_{t\le T}\mathrm{UpdateFLOPs}_t$
        (proxy $\Flops\times S_{\mathrm{tot}}$).
  \item \textbf{Constraint} --- holdout $\Acc$ / MSE within a
        tolerance of $\pi_0$ (e.g.\ $\Delta\Acc\ge -0.5\%$ abs).
\end{enumerate}
Inference latency is \emph{out of scope} (FWD stays on).
\end{definition}

\paragraph{What we optimize with PO metrics.}
Not ``higher PO is better.''
Higher $\PO_m$ means modality $m$ deserves \emph{more of a fixed
budget} (or others get frozen) so the same quality arrives with less
wasted BWD.

\paragraph{Brief loop (landing).}
\[
\boxed{
\begin{aligned}
&\text{1. Compute metrics on window }t
  \quad(\PO_m,\,\text{optional }\MMD_m,\mathrm{proto}_m,\Delta\PO_m)\\
&\text{2. Map}\to\alpha^{(t)}\text{ (or }w_i\text{)}
  \quad\texttt{metric\_to\_alpha}\ /\ \texttt{po\_iptw\_weights}\\
&\text{3. Actuate window }t{+}1\text{: }
  \{\lambda_m,\,s_m,\,\mathrm{freeze}_m,\,\pi^{\mathrm{stack}}\}\
  \text{or sample }w\\
&\text{4. Score speed: }\Flops,\ \sum s_m,\ \text{wall-clock};\
  \text{gate on }\Acc
\end{aligned}
}
\]
Causal rule: metrics at $t$ never reweight the same-window evaluation
labels used to report $\Acc_{t+1}$.

\section{Landing KPIs (report these, in this order)}
\label{sec:kpi}

\begin{center}
\small
\begin{tabular}{@{}llp{6.2cm}@{}}
\toprule
KPI & Role & Definition / procedure \\
\midrule
$\Flops$
  & \textbf{Primary speed}
  & $\#\{\text{mods with BWD on}\}/|\mathcal{M}|$
    averaged over post-train windows \\
$\bar s / S_{\mathrm{tot}}$
  & Primary speed
  & Fraction of step budget actually spent
    (after freeze / realloc) \\
Wall-clock$_{t\to t+1}$
  & Primary speed
  & Update time only (exclude sensor fit if amortized) \\
$T(\Acc^\star)$
  & Primary speed
  & Windows until holdout Acc hits bar \\
$\Delta\Acc$ vs equal
  & \textbf{Constraint}
  & Must stay $\ge -\varepsilon$; else metric landing fails \\
$\Delta$MSE vs equal
  & Constraint
  & Same; report with Acc \\
$H(\alpha)$
  & Diagnostic
  & Entropy of $\alpha$; mid preferred; not a win alone \\
Hard-rank / IPTW next-MSE
  & Path~A diagnostic
  & Only when sample reweight is the actuator \\
\bottomrule
\end{tabular}
\end{center}

\begin{remark}[Do not land Acc as the primary metric]
Acc lift without FLOPs / step / wall-clock is \emph{not} an
acceleration win.
Conversely, FLOPs cut that breaks the Acc constraint is a failed
landing---retune $\theta_{\mathrm{fr}}$ / $\tau$ / floor, do not ship.
\end{remark}

\section{Shared landing procedure (all metric versions)}
\label{sec:shared}

Code path: \texttt{metric\_to\_alpha} $\to$
\texttt{next\_step\_actuators} $\to$ optimizer on window $t{+}1$.

\paragraph{Procedure P0 (every window).}
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item \textbf{Sense.} Fit / reuse nuisances; record
        $\PO_m$ (required); optionally $\MMD_m$, proto distance,
        uni-acc, VIMP, $\PO_m^{\mathrm{prev}}$.
  \item \textbf{Map.} Choose version $v$ (Table~\ref{tab:land});
        $\alpha^{(t)}=\mathrm{Map}_v(\mathrm{sensors}_t)$.
  \item \textbf{Actuate (accelerate).}
        \begin{itemize}[leftmargin=1em,itemsep=1pt]
          \item $\lambda_m\leftarrow\texttt{alpha\_to\_lr}(\alpha;\beta)$
                --- put LR on high-$\PO$ mods.
          \item $s_m\leftarrow$ integer alloc $\propto\alpha$,
                $\sum s_m\approx S_{\mathrm{tot}}$ --- dump steps
                to the spike.
          \item $\mathrm{freeze}_m\leftarrow[\alpha_m<\theta_{\mathrm{fr}}]$
                (BWD off, FWD on) --- \emph{the} FLOPs cut.
          \item $\pi^{\mathrm{stack}}\leftarrow\alpha$ --- optional;
                does not save FLOPs by itself.
        \end{itemize}
  \item \textbf{Train} window $t{+}1$ under $(\lambda,s,\mathrm{freeze})$.
  \item \textbf{Score.} Log $\Flops$, steps used, wall-clock,
        $\Acc$, MSE. Mark landing \texttt{OK} iff
        $\Flops < 1$ (or $T(\Acc^\star)$ smaller) \textbf{and}
        $\Delta\Acc\ge-\varepsilon$.
\end{enumerate}

\paragraph{Path A add-on (sample IPTW, when reject).}
If OnlineRFPerm rejects batch $t$: set
$w_i\propto\sqrt{\PO_i}$ (mean-1, clipped), fit next model with
\texttt{sample\_weight}=$w$.
Landing KPIs: next-MSE / wall-clock of that fit; gate Acc/MSE vs
uniform. Default soft $\sqrt{\PO}$; always-on without gate is not
the acceleration recipe (calm packs regress).

\section{Per-metric landing (focus = speed actuators)}
\label{sec:per}

Each row: \emph{when to fire}, \emph{map}, \emph{which speed knob},
\emph{pass/fail}.

\begin{center}
\small
\begin{tabular}{@{}p{1.55cm}p{2.4cm}p{2.6cm}p{2.8cm}@{}}
\toprule
Version & Fire when & Map (brief) & Speed landing \\
\midrule
\texttt{equal}
  & Baseline / Acc saturated
  & $\alpha=1/M$
  & No cut; use to measure $\Acc^\star$ bar \\
\texttt{po\_soft}
  & Clear $\PO$ spike, $M\ge3$
  & Softmax$(\PO/\tau)$
  & LR+steps $\to$ top-$\alpha$; freeze tail \\
\texttt{po\_minus\_cov}
  & High MMD, flat concept
  & Softmax$(\PO-\lambda\MMD)$
  & \textbf{Do not} buy steps for pure $X$-shift;
    freeze cov-heavy mods \\
\texttt{po\_gated}
  & Noise vs drift unclear
  & Gate then Softmax
  & Primary FLOPs saver on Affec-like;
    skip noisy BWD \\
\texttt{po\_proto}
  & Centroid move + $\PO$
  & Softmax$(\PO(1+\mathrm{proto})-\lambda\MMD)$
  & Concentrate steps on true move;
    Acc constraint easier \\
\texttt{po\_delta}
  & Rising $\Delta\PO$
  & Softmax$(\mathrm{EMA}+\gamma\Delta)$
  & Reallocate \emph{before} Acc drop
    $\Rightarrow$ smaller $T(\Acc^\star)$ \\
\texttt{po\_budget}
  & Need always-on towers
  & Floor + Softmax
  & Soft LR only; \textbf{no} freeze
    (speed via realloc, not cut) \\
\texttt{po\_next}
  & Stable hist + $\Delta\PO$
  & Forecast next $\alpha$
  & Same actuators from forecast;
    earlier freeze/realloc \\
\bottomrule
\end{tabular}
\label{tab:land}
\end{center}

\paragraph{Landing checklist (operator).}
\begin{enumerate}[leftmargin=1.4em,itemsep=2pt]
  \item Pick $v$ from the table for the pack regime (not from Acc chase).
  \item Fix $S_{\mathrm{tot}}$, $\theta_{\mathrm{fr}}$, $\tau$, $\varepsilon$.
  \item Run $\ge$ one burn-in then score P0 on holdout windows.
  \item \textbf{Pass:} $\Flops\downarrow$ (or $T(\Acc^\star)\downarrow$)
        with $\Delta\Acc\ge-\varepsilon$.
  \item \textbf{Fail:} Acc broken $\to$ raise floor / lower
        $\theta_{\mathrm{fr}}$ / switch to \texttt{po\_budget};
        or FLOPs flat $\to$ raise $\theta_{\mathrm{fr}}$ / use
        \texttt{po\_gated} (only if $M\ge3$).
\end{enumerate}

\section{Actuator $\leftrightarrow$ acceleration (why each knob)}
\label{sec:knobs}

\begin{center}
\small
\begin{tabular}{@{}llp{5.8cm}@{}}
\toprule
Actuator & Speeds post-train by & Landing note \\
\midrule
$\mathrm{freeze}_m$ (BWD)
  & Cutting inactive tower backward
  & Main $\Flops$ lever; FWD stays on \\
$s_m$ step alloc
  & Same wall-clock budget, more steps on spike
  & Must be wired to real optimizer steps \\
$\lambda_m$ LR mult
  & Faster correction on high-$\PO$ mods
  & Alone rarely cuts FLOPs; pair with freeze \\
$w_i\propto\sqrt{\PO}$
  & Fewer wasted steps on easy rows (Path~A)
  & Gated; report next-MSE + fit time \\
$\pi^{\mathrm{stack}}=\alpha$
  & Faster mixer convergence (indirect)
  & Secondary; do not count as FLOPs win alone \\
\bottomrule
\end{tabular}
\end{center}

\begin{remark}[Wire \texttt{step\_alloc} for real]
If the trainer only scales LR and ignores $s_m$, the landing loses
the largest unused acceleration lever.
Procedure: set per-mod max\_steps / grad accum from $s_m$ before
claiming speedups.
\end{remark}

\section{Smoke reading under the acceleration lens}
\label{sec:emp}

Source: \texttt{AGOD\_po\_risk\_train\_compare.md}.

\paragraph{Affec ($M{=}5$) --- acceleration headroom.}
\texttt{po\_gated}: $\Flops\approx 0.70$, Acc $\approx$ flat
$\Rightarrow$ \textbf{pass} as FLOPs landing.
\texttt{po\_proto}: small Acc lift at $\Flops\approx 0.825$
$\Rightarrow$ pass if bar is Acc-first with mild cut.
Prefer these over Acc-only ablations.

\paragraph{COCO / Fashion / Food (2-mod, Acc already high).}
$\Flops\approx 1$ for all versions at default $\theta_{\mathrm{fr}}$
$\Rightarrow$ \textbf{no acceleration claim}.
Landing: keep \texttt{equal}/\texttt{po\_budget}; do not freeze;
optional stack prior only.

\paragraph{Path~A stream packs.}
Gated $\sqrt{\PO}$ beats always-$\sqrt{}$ on update quality/cost
tradeoff ($6/6$ vs always); use gate so calm windows stay uniform
(zero extra weight compute / no harmful upweight).

%======================================================================
\section{Takeaway}
\label{sec:take}

\begin{enumerate}[leftmargin=1.4em,itemsep=3pt]
  \item \textbf{Goal:} accelerate post-training
        ($\Flops$, steps, wall-clock, $T(\Acc^\star)$);
        Acc is a constraint.
  \item \textbf{Metrics land only via actuators:}
        sense $\to$ map $\to$ freeze/LR/steps/IPTW $\to$ speed KPIs.
  \item \textbf{Procedure P0} is the shared landing;
        Table~\ref{tab:land} chooses which map to fire.
  \item \textbf{Ship rule:} FLOPs or $T(\Acc^\star)$ down
        \emph{and} $\Delta\Acc\ge-\varepsilon$; else retune, don't claim.
  \item \textbf{Next:} wire real \texttt{step\_alloc};
        Pareto sweep $\theta_{\mathrm{fr}}$ on Acc@FLOPs.
\end{enumerate}
