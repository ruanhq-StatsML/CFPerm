\section{Product next-step logic (straightforward)}
%----------------------------------------------------------------------

\begin{enumerate}[leftmargin=1.3em,itemsep=0.35em]
  \item Fit $\varphi$, $\hat\tau$ (PermuCATE RF or CausalForestDML).
  \item Unit risk $u=(\varphi-\hat\tau)^2$.
  \item Mag$\times$Var: $\mu=\mathbb{E}[u]$, $s=\mathrm{Std}$(sub-env mean $u$).
  \item $\mathrm{Var}(u)$: \textbf{CF leaf-size-debiased + leaf merge}
        (product default).
  \item $Y$-flip on text integrity boards (attacked rows only).
  \item \texttt{triage()} $\to$ hold / conformal / VAI\_freeze.
  \item Honesty: $\mathrm{bias\_ratio}=B_{\mathrm{leaf}}/\widehat V_{\mathrm{naive}}$.
\end{enumerate}

\paragraph{Gate (who freezes).}
Primary sensors $=$ Mag$\times$Var $(\mu,s,J)$ + $Y$-flip.
$\mathrm{Var}(u)$ is \emph{confirmation / honesty}, not a solo absolute
threshold (that false-froze benign covariate / synonym boards).

%----------------------------------------------------------------------
\section{Why Var(u) (statistical insight)}
%----------------------------------------------------------------------

$u_i=r_i^2$ with $r_i=\varphi_i-\hat\tau(X_i)$ is heavy-tailed;
$\mathrm{Var}(u)=m_4-m_2^2$ is unstable if taken as a raw residual moment.
Obtain $\mathrm{Var}(u)$ from the forest / Bayesian uncertainty of
$\hat\tau$, and \textbf{debias leaf-size heterogeneity} in
$\mathrm{Var}(\hat\tau)$ --- otherwise upward bias masquerades as malignant.

\[
\widehat V_{\mathrm{debiased}}
=\widehat V_{\mathrm{naive}}-\widehat B_{\mathrm{leaf}},
\quad
\widehat B_{\mathrm{leaf}}
=\frac{B-1}{B^2}\sum_b\frac{\hat\sigma^2_{b,i}}{n_{b,i}}.
\]
Leaf-wise merge of $u$ borrows strength inside the leaf before aggregating.
