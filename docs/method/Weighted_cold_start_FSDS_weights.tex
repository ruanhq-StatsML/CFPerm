\begin{abstract}
\noindent
\textbf{Two questions, two tools.}
(A)~Which \emph{sample-level} weight has power on the new category?
$\to$ answer with \textbf{complexity probes} (VIMP-top, gap, hard-prec).
(B)~Is that weight's \textbf{novel SGD share} enough, or do we still need
downsample?
$\to$ answer with $s(w)=\sum w_i N_i/\sum w_i$.
Complexity selects the channel; $s(w)$ audits mass vs 1:1 downsample.
Neither replaces the other.
Headline: best sample-weight mean $\Delta$novel $+0.178$;
vs \texttt{ds\_known} incremental $\mathbf{+0.052}$ ($5/9$).
Default: no label-downsample.
\end{abstract}

%======================================================================
\section{Setup (sample-level only)}
%======================================================================

\begin{itemize}
  \item \textbf{Feature-level (FSDS):} VIMP $\in\mathbb{R}^p$ picks subspace
        $\mathcal{S}$. Not a training weight.
  \item \textbf{Sample-level (Layer~2):} $w\in\mathbb{R}^{n_{\mathrm{LIVE}}}$
        in \texttt{partial\_fit(..., sample\_weight=$w$)}.
        Soft / hard / fused live here.
\end{itemize}
\[
w^{\mathrm{sample}}_i=1+\lambda m_i,\quad
w^{\mathrm{subset}}_i\ \mathrm{boosts}\ \{m_i\ge q\},\quad
w^{\mathrm{fused}}=w^{\mathrm{sample}}\odot w^{\mathrm{subset}}.
\]
KPI $=$ novel acc. Optional private-live fuse on $m$ before building $w$.

%======================================================================
\section{Logic A --- Complexity $\to$ weight power
(always needed to pick the channel)}
%======================================================================

\paragraph{What ``power'' means.}
Power $=$ how much a sample-level channel lifts novel acc vs uniform.
Complexity $=$ whether the novel support is recoverable in $\mathcal{S}$:
$n_{\mathrm{nov}}$, gap, $b/w$, hard-precision, VIMP-top --- not label count alone.

\begin{center}
\scriptsize
\begin{tabular}{@{}llcc@{}}
\toprule
Complexity regime & Read & Channel with power & Evidence \\
\midrule
Simple blob
  & gap $\gtrsim 0.6$, small $n_{\mathrm{nov}}$
  & \textbf{fused} (or hard)
  & wine $+0.33$, cancer $+0.20$ \\
Family blob
  & $b/w\gg 1$, $E[m\mid\mathrm{nov}]\approx 1$
  & \textbf{hard} (not fused)
  & HAR $+0.32$ (fus $-0.05$) \\
Messy / high-$d$
  & gap $\lesssim 0.2$ or $<0$
  & \textbf{soft $+$ fuse}
  & news20b $+0.25$, jvowel $+0.11$ \\
Fragmented
  & large $n_{\mathrm{nov}}$, VIMP $\lesssim 0.10$
  & \textbf{refuse}
  & ISOLET $0$ \\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Why this logic is required.}
Without complexity probes you cannot choose among soft / hard / fused / refuse:
the same $w^{\mathrm{fused}}$ helps wine and hurts HAR.
Anchors:
$\mathrm{corr}(n_{\mathrm{nov}},\mathrm{fused\ power})\approx -0.54$,
$\mathrm{corr}(v(\mathcal{S}),\mathrm{fused})\approx +0.51$,
$\mathrm{corr}(\mathrm{gap},\mathrm{soft\ power})\approx -0.32$,
$\mathrm{corr}(\mathrm{H\text{-}prec},\mathrm{hard\ power})\approx +0.41$.

\paragraph{Ops rule (from A).}
VIMP$\lesssim 0.10\to$refuse;\ 
$E[m\mid\mathrm{nov}]\gtrsim 0.9$ \& hard$\ge$fused$\to$hard;\ 
gap$\lesssim 0.25\to$soft$+$fuse;\ 
else$\to$fused.

%======================================================================
\section{Logic B --- Novel SGD share $s(w)$
(needed when asking about downsample)}
%======================================================================

\paragraph{Definition (sample-level, verified).}
\[
s(w)
=\frac{\sum_i w_i N_i}{\sum_i w_i}
=\frac{n_N\,\bar w_N}{n_N\,\bar w_N+n_K\,\bar w_K}.
\]
Also track $\bar w_N/\bar w_K$.
\texttt{ds\_known} forces $s=\tfrac12$ by discarding $\sim$75\% known rows.
Uniform has $s\approx 0.17$--$0.23$.

\paragraph{When is $s(w)$ needed?}
\begin{itemize}
  \item \textbf{Needed} to decide downsample: does the chosen sample-weight
        already buy enough novel SGD mass, or must we force $s=\tfrac12$ by
        dropping known?
  \item \textbf{Not needed} to pick soft vs hard vs fused --- that is Logic~A.
        Empirically $\mathrm{corr}(s^\star,\Delta\mathrm{novel}_w)\approx 0.05$:
        mass alone does not rank channels.
\end{itemize}

\begin{center}
\scriptsize
\begin{tabular}{@{}lcccc@{}}
\toprule
Board & channel (from A) & $s(w)$ & $\bar w_N/\bar w_K$ & Downsample? \\
\midrule
wine / mice & fused & $0.46$--$0.49\approx\tfrac12$ & $3.2$--$3.5$
  & \textbf{No} --- mass-matched \\
har / cancer / cnae9 / news / jvowel
  & hard or soft & $0.28$--$0.37<\tfrac12$ & $1.6$--$2.2$
  & \textbf{No} --- quality beats blunt $1$:$1$ \\
covtype & soft & $0.25\ll\tfrac12$ & $1.7$
  & \textbf{Yes} --- then soft on kept \\
isolet & (refuse) & $0.43\approx\tfrac12$ & $3.0$
  & Irrelevant --- subspace bad \\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Four readings of $s(w)$.}
\textbf{J1} $s\approx\tfrac12$ $\Rightarrow$ novel ties ds, keep all known (wine/mice).
\textbf{J2} $s<\tfrac12$ but $m$ quality high $\Rightarrow$ $W>\mathrm{ds}$
(har $+0.14$ incr; news $+0.25$; jvowel ds known $-0.22$ vs soft known $+0.04$).
\textbf{J3} $s$ stuck $\sim 0.25$ $\Rightarrow$ ds supplies mass, soft ranks inside
(covtype; combo $+0.14$ over ds alone).
\textbf{J4} high $s$, tiny VIMP $\Rightarrow$ mass$\neq$quality (ISOLET).

\paragraph{Ablation anchors.}
best sample-weight $-$ uniform: $+0.178$;\
\texttt{ds\_known}$-$uniform: $+0.127$;\
\texttt{ds\_random}$-$uniform: $-0.023$ (not smaller $n$);\
best weight $-$ \texttt{ds\_known}: $\mathbf{+0.052}$ ($5/9$).

\section{How A and B fit together}
%======================================================================

\begin{center}
\small
\begin{tabular}{@{}lll@{}}
\toprule
Step & Question & Tool \\
\midrule
1 & Which sample-level channel has power?
  & \textbf{A:} complexity (gap, H-prec, VIMP) \\
2 & Build that $w$ (soft / hard / fused / refuse)
  & Layer~2 formulas \\
3 & Is novel SGD share enough vs blunt rebalance?
  & \textbf{B:} $s(w)$ vs $\tfrac12$ \\
4 & If $s(w)\ll\tfrac12$ and valid.\ ds helps
  & ds\_known, then soft on remainder \\
5 & If VIMP diffuse
  & refuse both weights and ds-as-fix \\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Straightforward claim.}
Complexity grants the \emph{right} to use a weight channel (power).
Novel SGD share audits whether that right already delivers enough
\emph{mass} --- or whether downsample is still on the table.
You always run~A; you run~B when the product question is ``weights or
downsample?''.

Never random-downsample.

%======================================================================
\section{Adaptation opportunity $O_t$
(gate for attribute + separate)}
%======================================================================

Next batch is unknown. So we do not ask only ``did sep work on this
batch?''; we ask whether there is \emph{opportunity to adapt to new
categories now}:
\[
O^{\mathrm{local}}_t=(D_t L_t R_t M_t)^{1/4},\quad
O^{\mathrm{persist}}_t=O^{\mathrm{local}}_t\cdot\mathrm{Stab}_t.
\]
$D$=error-rise detect, $L$=VIMP mass on $\mathcal{S}$,
$R$=room to push off known prototypes, $M$=actable high-mem mass,
$\mathrm{Stab}$=footprint overlap vs last batch (persist discount).
Policy: $O^{\mathrm{local}}\ge\tau_{\mathrm{sep}}\Rightarrow$ separate;
mass$+O$ high $\Rightarrow$ expand with sample weights;
$O^{\mathrm{persist}}$ low $\Rightarrow$ drop/replace $\mathcal{S}$-offset.
Track benches: C-GCD (CIFAR-100 / IN-100 / Tiny / CUB; Old/New/All);
PDA/OSDA (Office-Home / VisDA / DomainNet; H-score).

%======================================================================
\section{Pipeline v3 --- representation before the head}
%======================================================================

Weights answer \emph{whom to emphasize at expand}.
When novel is sparse, updating the head first overfits and pulls old
boundaries. Order matters --- and each step is gated by $O_t$:

\begin{enumerate}
  \item \textbf{Detect + attribute.}
        Error-rise gate; FSDS RF-domain VIMP on $W\in\{0,1\}$
        $\to$ subspace $\mathcal{S}$ and membership $m_i$.
        Role $=$ focus, not classify.
  \item \textbf{Representation separation (instant learn).}
        Known prototypes $\mu_c$ on $\mathcal{S}$.
        High-$m_i$ LIVE rows with
        $d_i=\min_c\|x_i^{\mathcal{S}}-\mu_c\|<\mathrm{margin}$
        get a margin push away from the nearest old center.
        Persist a mean $\mathcal{S}$-offset for inference.
        Sparse samples suffice: we only stop collapse into old centers.
  \item \textbf{Dynamic head expand (learn completion).}
        Buffer novel / high-$m$ rows. Expand when
        $\mathrm{buffer}\ge N^\star$ \emph{or} the attributed footprint
        stabilizes. Then joint-fit with novel buffer $+$ known replay,
        using the sample-level channel from Logic~A.
  \item \textbf{Optional TTA on $\mathcal{S}$.}
        Entropy-min only on parameters tied to $\mathcal{S}$
        (tabular: $\mathrm{coef}_{[:,\mathcal{S}]}$); freeze the rest.
        Revert if entropy rises.
\end{enumerate}

\paragraph{Rule of thumb.}
Separate first, classify second.
Sample-level $w$ and $s(w)$ still audit the expand step;
they do not replace the geometric preconditioner on $\mathcal{S}$.

%======================================================================
\section{Takeaway}
%======================================================================

\begin{enumerate}
  \item \textbf{Weight power} $=$ sample-level soft/hard/fused lift, chosen by
        complexity. Always needed \emph{at expand}.
  \item \textbf{Novel SGD share} $s(w)$ $=$ mass audit vs 1:1 downsample.
        Needed for the downsample decision; not for channel choice
        ($\mathrm{corr}(s,\Delta\mathrm{novel})\approx 0.05$).
  \item \textbf{Repr-sep} $=$ instant learn on $\mathcal{S}$ before the head.
        Expand completes learning once buffer / footprint is ready.
  \item Together: attribute $\to$ separate $\to$ expand($w$) $\to$ optional TTA.
        Incr of best $w$ over \texttt{ds\_known} $=+0.052$.
\end{enumerate}
