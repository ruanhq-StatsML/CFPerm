\begin{abstract}
\noindent
\textbf{Claim.}
Under category shift, FSDS-localized \emph{sample-level} weights cold-start
new categories better than uniform \texttt{partial\_fit}, and usually better
than blunt known-class downsampling.
\textbf{How.}
RF-domain VIMP $\to$ membership $m$ $\to$ soft / hard / fused observation
weights, chosen by novel-category complexity.
\textbf{Headline numbers (9 boards).}
best weight mean $\Delta$novel $+0.178$ vs uniform;
\texttt{ds\_known} $+0.127$;
\textbf{incremental} best-weight $-$ \texttt{ds\_known} $\mathbf{+0.052}$
(win $5/9$).
Random downsample $-0.023$ --- gain is selective mass, not smaller $n$.
Default: \textbf{do not} label-downsample; ship soft/hard/fused by probes;
downsample only when $s(w)$ cannot approach $1/2$ (covtype-like).
\end{abstract}

%======================================================================
\section{Pipeline (locked)}
%======================================================================

\begin{enumerate}
  \item \textbf{Detect} LIVE/REF error rise.
  \item \textbf{Layer 1 (FSDS):} RF on $W=\mathbf{1}\{\mathrm{LIVE}\}$
        $\to$ VIMP $\to$ top-$K$ subspace $\mathcal{S}$.
  \item \textbf{Localize:} $m_i=$ clipped z-distance to REF in $\mathcal{S}$.
        Optional fuse $m\leftarrow\max(m,\,1\{y\in\mathrm{private\text{-}live}\})$.
  \item \textbf{Layer 2 (sample weights):}
        $w^{\mathrm{sample}}=1+\lambda m$,\
        $w^{\mathrm{subset}}$ boosts $\{m\ge q\}$,\
        $w^{\mathrm{fused}}=w^{\mathrm{sample}}\odot w^{\mathrm{subset}}$.
  \item \textbf{Select channel} by complexity probes
        (VIMP-top, gap, hard-prec, $E[m\mid\mathrm{nov}]$)
        $\in\{\mathrm{refuse},\mathrm{soft},\mathrm{hard},\mathrm{fused}\}$.
  \item \textbf{Adapt} \texttt{partial\_fit}; KPI $=$ \textbf{novel acc}
        (known/forgetting $=$ cost).
\end{enumerate}

\paragraph{Diagnostic.}
$\mathrm{gap}=\mathbb{E}[m\mid\mathrm{nov}]-\mathbb{E}[m\mid\mathrm{kn}]$;
novel mass share $s(w)=\sum w_i N_i/\sum w_i$ ($N=$ novel indicator).

%======================================================================
\section{Novel-category complexity $\leftrightarrow$ weight power}
%======================================================================

Complexity is not ``how many new labels'' alone.
It is whether the novel support is a \emph{recoverable target} in $\mathcal{S}$:
$n_{\mathrm{nov}}$, within-spread, $b/w$, gap, hard-precision, ambient $p$.

\begin{center}
\scriptsize
\begin{tabular}{@{}lcccccc@{}}
\toprule
Board & $n_{\mathrm{nov}}$ & gap & $b/w$ & H-prec
  & soft power & hard / fused \\
\midrule
wine    & 1  & $+0.71$ & 7.2  & 0.48 & $0$ & fused $\mathbf{+0.33}$ \\
cancer  & 1  & $+0.64$ & 7.7  & 0.46 & $0$ & hard/fus $\mathbf{+0.20}$ \\
har     & 3  & $+0.79$ & 54.5 & 0.50 & $+0.05$ & hard $\mathbf{+0.32}$ / fus $-0.05$ \\
cnae9   & 4  & $+0.61$ & 1.4  & 0.47 & $\mathbf{+0.19}$ & $=+0.19$ \\
mice    & 4  & $+0.36$ & 1.5  & 0.40 & $+0.06$ & fused $\mathbf{+0.17}$ \\
news20b & 2  & $+0.18$ & 0.5  & 0.26 & $\mathbf{+0.25}$ & $+0.12$ \\
covtype & 2  & $+0.15$ & 1.7  & 0.22 & $\mathbf{+0.10}$ & fus $0$ \\
jvowel  & 4  & $-0.04$ & 1.0  & 0.18 & $\mathbf{+0.11}$ & $+0.05$ \\
isolet  & 13 & $+0.29$ & 1.3  & 0.33 & $-0.05$ & $0$/$-0.05$ \\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Soft.}
Continuous tilt; wins when alignment is imperfect
(news20b $+0.25$, jvowel $+0.11$, covtype $+0.10$).
$\mathrm{corr}(\mathrm{gap},\mathrm{soft\ power})\approx -0.32$:
soft is for hard-to-align complexity.

\paragraph{Hard.}
Needs saturated novel cluster ($E[m\mid\mathrm{nov}]\approx 1$).
HAR: three posture labels look complex, but $b/w=54.5$ makes one
\emph{family} vs locomotion --- hard $+0.32$; fused over-concentrates ($-0.05$).
Power tracks \emph{cluster} complexity, not label count
($\mathrm{corr}(\mathrm{H\text{-}prec},\mathrm{hard\ power})\approx +0.41$).

\paragraph{Fused.}
Needs sharp VIMP $+$ aligned $m$.
wine $+0.33$, mice $+0.17$, cnae9 $+0.19$.
$\mathrm{corr}(n_{\mathrm{nov}},\mathrm{fused\ power})\approx -0.54$,
$\mathrm{corr}(v(\mathcal{S}),\mathrm{fused})\approx +0.51$.
Many heterogeneous novel labels $+$ diffuse VIMP (ISOLET) $\Rightarrow$ power $\approx 0$.

\paragraph{Power law.}
Simple blob $\to$ fused/hard;\ 
messy/high-$d$ $\to$ soft;\ 
fragmented$+$diffuse VIMP $\to$ refuse.

%======================================================================
\section{Post-FSDS selection (ops rule)}
%======================================================================

\begin{enumerate}
  \item $v(\mathcal{S})\lesssim 0.10$ $\to$ \textbf{refuse} (ISOLET).
  \item $E[m\mid\mathrm{nov}]\gtrsim 0.9$, H-prec $\gtrsim 0.35$, hard $\ge$ fused
        $\to$ \textbf{hard subset} (HAR, cancer).
  \item gap $\lesssim 0.25$ or weak $m$--novel corr
        $\to$ \textbf{soft $+$ private-live fuse} (jvowel, news20b, covtype).
  \item Else $\to$ \textbf{fused} (wine, mice, cnae9).
\end{enumerate}
Fuse is a complexity patch when geometry under-covers novel
(jvowel gap$<0$; mice product $-0.39\to$fused $+0.17$).

\subsection{Gradient-mass account (why)}

\[
s(w)=\frac{\sum w_i N_i}{\sum w_i},\qquad
\Delta\mathrm{novel}(w)
\;\approx\;
f\big(s(w),\,\mathrm{quality}(m)\big)
\;-\;
c(\mathrm{known\ discard}).
\]

\paragraph{Default.}
No label-downsample. Ship post-FSDS channel by the card.
Never random-downsample.

%======================================================================
\section{Takeaway}
%======================================================================

\begin{enumerate}
  \item Sample-level weights work when the new category (or novel \emph{family})
        is recoverable in the FSDS subspace; channel follows complexity.
  \item Soft handles weak alignment; hard handles saturated clusters;
        fused needs sharp$+$aligned; refuse when fragmented$+$diffuse.
  \item Downsample is not default: incremental best-weight over
        \texttt{ds\_known} is $\mathbf{+0.052}$ ($5/9$).
        Use ds only when mass share cannot approach $1/2$, then soft-weight.
  \item Peaks: soft $+0.25$, hard $+0.32$, fused $+0.33$;
        best-w mean $+0.178$; ds mean $+0.127$.
\end{enumerate}
