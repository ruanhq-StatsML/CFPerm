\begin{abstract}
\noindent\textbf{Objective:} cold-start the \emph{new class} when category shift
arrives sparsely --- without drowning \texttt{partial\_fit} in known-class
recirculation.
\textbf{How:} two weight layers.
(1)~\emph{FSDS feature-level} --- RF-domain VIMP localizes which features
moved LIVE vs REF.
(2)~\emph{Post-localization} --- membership on that subspace becomes soft
\texttt{sample\_w} and hard \texttt{subset\_w}; the product
$w^{\mathrm{fused}}=w^{\mathrm{sample}}\odot w^{\mathrm{subset}}$ is the
default cold-start channel.
\textbf{Headline:} mean $\Delta$ novel $\approx{+}0.165$ vs uniform on four
boards; gains sit on the new class.
\end{abstract}

%----------------------------------------------------------------------
\section{PO logic in one page}
%----------------------------------------------------------------------

\paragraph{Problem.}
Standard UDA assumes a shared label set.
Category shift (C-GCD / PDA) injects private-live labels.
The product question is not only ``did shift fire?'' but
\textbf{can we cold-start the new class?}

\paragraph{Two weight layers (locked).}
\begin{center}
\small
\begin{tabular}{@{}llp{0.48\textwidth}@{}}
\toprule
Layer & Object & What it does \\
\midrule
1. Feature-level
  & FSDS RF-domain VIMP
  & Attributes LIVE$\leftrightarrow$REF on $W\in\{0,1\}$;
    top-$K$ features = subspace for localization;
    mild amplify of those coords at adapt. \\
2. After localization
  & Observation weights
  & Membership $m_i$ in the VIMP subspace
    $\to$ soft $w^{\mathrm{sample}}_i=1+\lambda m_i$
    and hard $w^{\mathrm{subset}}_i$ on $\{m_i\ge q\}$;
    default $w^{\mathrm{fused}}=w^{\mathrm{sample}}\odot w^{\mathrm{subset}}$. \\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Claim for PO.}
Layer~1 diagnoses \emph{where} (which features / subspace).
Layer~2 decides \emph{whom to upweight} (which rows look new-category-like).
Membership alone is diagnostic; fused observation weights are the adaptation
lever. Weights are gated by label-support Jaccard (PDA incomparable $\Rightarrow$
no directed update) and tempered when $m$ is diffuse.

%----------------------------------------------------------------------
\section{Pipeline}
%----------------------------------------------------------------------

\begin{enumerate}
  \item \textbf{Detect:} LIVE/REF error-rise gate (OnlineRFPerm-style).
  \item \textbf{Layer 1 --- attribute:} RF-domain VIMP on batch $W\in\{0,1\}$;
        top-$K$ feature subspace.
  \item \textbf{Localize:} z-score distance in that subspace
        $\to$ membership $m_i\in[0,1]$.
  \item \textbf{Boundary:} Jaccard(label supports)${<}0.2$ $\Rightarrow$
        \texttt{incomparable} (no directed adapt).
  \item \textbf{Fuse labels:} $m_i\leftarrow\max(m_i,\,\mathbf{1}\{y_i\in\mathrm{private\text{-}live}\})$.
  \item \textbf{Layer 2 --- weights:}
        $w^{\mathrm{sample}}_i=1+\lambda_{\mathrm{eff}} m_i$ (soft);
        $w^{\mathrm{subset}}_i$ boosts $\{i:m_i\ge q\}$ (hard);
        default $w^{\mathrm{fused}}=w^{\mathrm{sample}}\odot w^{\mathrm{subset}}$,
        tempered when $m$ is diffuse.
  \item \textbf{Adapt:} mix weighted LIVE with known REF replay ($w{=}1$);
        \texttt{partial\_fit} with $w^{\mathrm{fused}}$
        (+ mild amplify of Layer-1 features);
        \texttt{auto} falls back to soft-only when concentration is low.
\end{enumerate}

\paragraph{Auto policy.}
Sharp footprint $\to$ fused; diffuse footprint $\to$ sample-only.
Guards covertype-style boards where stacking hard subset can hurt novel acc.

\paragraph{Channel ablation (mean $\Delta$ novel vs.\ uniform).}
\texttt{sample\_w} alone $\approx{-}0.007$;
\texttt{subset\_w} alone $\approx{+}0.031$;
\texttt{sample}$\times$\texttt{subset} $\approx{+}0.117$;
\texttt{both\_fused} $\approx{+}0.165$.
Layer~2 soft $\times$ hard is what moves the new class;
Layer~1 supplies the subspace that makes membership meaningful.

\paragraph{Reading by board.}
\begin{itemize}
  \item \textbf{wine:} novel $2/3\to 1$; known flat --- ideal cold-start.
  \item \textbf{cancer:} novel to $1.0$; known dips slightly --- acceptable tradeoff.
  \item \textbf{news20b:} novel doubles ($0.125\to0.25$;
        \texttt{sample\_w\_fused} reaches $0.375$),
        but known/forgetting pay --- high-d text is expensive.
  \item \textbf{covtype:} fused does not lift novel; soft
        \texttt{sample\_w\_fused} reaches $0.619$ ($+0.095$) ---
        prefer auto/sample when concentration is moderate.
\end{itemize}

%----------------------------------------------------------------------
\section{Takeaway for PO}
%----------------------------------------------------------------------

\begin{enumerate}
  \item \textbf{Layer 1 (FSDS feature-level)} localizes the shift subspace.
  \item \textbf{Layer 2 (post-localization)} upweights new-category-like rows
        via $w^{\mathrm{sample}}\times w^{\mathrm{subset}}$.
  \item Product metric $=$ novel acc (+ known retention / boundary honesty).
  \item Fused is default when the footprint is sharp; otherwise soft sample\_w.
  \item PDA: low label Jaccard $\Rightarrow$ incomparable, no directed update.
\end{enumerate}
