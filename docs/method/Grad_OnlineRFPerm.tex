\begin{abstract}
\textbf{One-line conclusion.}
It can detect the freezing point earlier than the other procedures.

We monitor distribution / concept shift through the \emph{serving}
gradient energy of a frozen reference network, rather than through
serving MSE alone. The test statistic is a \textbf{single} scalar
$\| \gradU L \|_2$ over all \emph{unfrozen} parameters.
One continuous-time OnlineRFPerm stream (rank / EWMA $p$ + online FDR)
yields a global reject. Per-layer gradient shares are diagnostics only
and do \emph{not} receive separate hypothesis tests---avoiding
cross-parameter multiple testing.

On five MVP packs $\times$ five seeds, Grad rejects before
serving-MSE OnlineRFPerm by about $3.2$ batches on average
($72\%$ earlier). A Sep18 add-on on two OpenML packs
(\texttt{adult}, \texttt{nomao}) repeats the same protocol and
yields mean $\Lead\approx -3.6$ ($80\%$ earlier; $100\%$ not later).
Pooled over all $35$ runs: mean $\Lead\approx -3.34$,
$P(\Lead<0)\approx 74\%$. Supplementary experiments cover baseline
alarm rate / grace, $\alpha$-sensitivity, a freeze closed-loop
(MSE vs.\ FLOPs), and extra stream packs (metro / beijing / stocks / waymo).
\end{abstract}

\section{Setup}
\label{sec:setup}

Let $f_{\mathrm{ref}}(\,\cdot\,;\theta)$ be a reference predictor
pretrained on a burn-in regime and then \textbf{frozen}
(weights are not stepped during monitoring).
At stream time $t=0,1,2,\ldots$ we observe a batch
$(X_t,Y_t)$ and a serving loss
$L_t(\theta)=L\bigl(f_{\mathrm{ref}}(X_t;\theta),Y_t\bigr)$
(e.g.\ batch MSE).

\begin{definition}[Unfrozen parameter set]
\begin{equation}
  \theta_U
  \;=\;
  \bigl\{\,
    \theta_i \in \theta
    \;\big|\;
    \texttt{requires\_grad}(\theta_i)=\mathrm{true}
  \,\bigr\}.
\end{equation}
Frozen / stopped parameters are excluded from the statistic.
\end{definition}

A single backward pass on $L_t$ yields
$\nabla_{\theta_i} L_t$ for each $\theta_i\in\theta_U$.

%======================================================================
\section{Scalar gradient energy (the gate)}
\label{sec:scalar}

\begin{definition}[Global unfrozen grad energy]
Concatenate all unfrozen grads into one Euclidean vector:
\begin{equation}
\label{eq:gt}
  g_t
  \;=\;
  \bigl\| \gradU L_t \bigr\|_2
  \;=\;
  \Biggl(
    \sum_{\theta_i \in \theta_U}
      \bigl\| \nabla_{\theta_i} L_t \bigr\|_2^{2}
  \Biggr)^{1/2}.
\end{equation}
\end{definition}

\begin{remark}[What $g_t$ is \emph{not}]
\label{rem:not}
\begin{enumerate}[leftmargin=*,itemsep=0.25em]
  \item \textbf{Not} a mean of per-layer / per-parameter norms:
  \[
    g_t
    \;\neq\;
    \frac{1}{|\theta_U|}
    \sum_{i}
      \bigl\| \nabla_{\theta_i} L_t \bigr\|_2,
  \]
  nor a mean of ``normalized'' layer norms.
  \item \textbf{Not} a family of per-layer OnlineRFPerm tests whose
  rejects are OR-ed together (cross-parameter multiple testing).
\end{enumerate}
Equation~\eqref{eq:gt} is one scalar $\Rightarrow$ one hypothesis stream.
\end{remark}

%======================================================================
\section{Continuous-time OnlineRFPerm on $g_t$}
\label{sec:rfperm}

Let $b$ be the burn-in length. During burn-in we estimate
\begin{equation}
\label{eq:eref}
  e_{\mathrm{ref}}
  \;=\;
  \frac{1}{b}\sum_{s=0}^{b-1} g_s
\end{equation}
and accumulate history without rejecting.

\begin{definition}[Grad test statistic]
For $t\ge b$,
\begin{equation}
\label{eq:Tt}
  T_t \;=\; g_t - e_{\mathrm{ref}}.
\end{equation}
Large unexpected gradient energy $\Rightarrow$ large $T_t$
(same orientation as MSE-OnlineRFPerm).
\end{definition}

\begin{definition}[Rank / EWMA $p$-value]
Let $\mathcal{H}_{t}=\{T_s:s<t\}$. Uniform rank:
\begin{equation}
\label{eq:prank}
  p_t^{\mathrm{rank}}
  \;=\;
  \frac{1 + \#\{\, T\in\mathcal{H}_{t}: T_t \le T \,\}}
       {|\mathcal{H}_{t}| + 1}.
\end{equation}
EWMA (default), ages $a(T)=0$ for the newest entry,
weights $w(T)=e^{-\lambda a(T)}$:
\begin{equation}
\label{eq:pewma}
  p_t^{\mathrm{EWMA}}
  \;=\;
  \frac{\varepsilon + \sum_{T\in\mathcal{H}_{t}} w(T)\,
        \mathbf{1}\{T_t \le T\}}
       {\varepsilon + \sum_{T\in\mathcal{H}_{t}} w(T)},
  \qquad \varepsilon\downarrow 0.
\end{equation}
Large $T_t$ $\Rightarrow$ small $p_t$.
\end{definition}

\begin{definition}[Online FDR (alpha-investing)]
Wealth $W_0=1$. At $t\ge b$ spend
\begin{equation}
\label{eq:alphat}
  \alpha_t \;=\; W_t \cdot \frac{\alpha}{1+\alpha},
\end{equation}
reject if $p_t < \alpha_t$, and update
\begin{equation}
\label{eq:wealth}
  W_{t+1}
  \;=\;
  \begin{cases}
    W_t + \alpha, & \text{if reject},\\[0.25em]
    \max(W_t - \alpha_t,\,\varepsilon), & \text{otherwise}.
  \end{cases}
\end{equation}
\end{definition}

Global Grad reject:
\begin{equation}
\label{eq:Rt}
  R_t^{\mathrm{grad}}
  \;=\;
  \mathbf{1}\{ p_t < \alpha_t \}.
\end{equation}
\textbf{Exactly one such stream}---no layer-wise FDR.

\paragraph{Optional grace.}
To damp burn-edge noise, ignore rejects on
$t\in\{b,\ldots,b+g-1\}$ when reporting first-reject time
(grace window $g$; does not change the underlying $p_t$).

\begin{definition}[Baseline alarm rate]
Over a window of batches $\mathcal{W}$ (default: post-burn, or post-burn+grace),
\begin{equation}
\label{eq:AR}
  \mathrm{AR}
  \;=\;
  \frac{n_{\mathrm{alarm}}}{n_{\mathrm{batch}}}
  \;=\;
  \frac{\#\{ t\in\mathcal{W}: R_t=1 \}}{|\mathcal{W}|}.
\end{equation}
This is a \textbf{baseline reference reject frequency}, \emph{not} a Type-I
false-alarm rate: with ongoing parameter / stream updates one cannot
rigorously claim a stationary DGP, so a moderately high AR is expected
and acceptable as a baseline.
\end{definition}

%======================================================================
\section{Layer shares (diagnostic only)}
\label{sec:shares}

Partition $\theta_U$ into layers $\ell\in\mathcal{L}$:
\begin{equation}
\label{eq:gl}
  g_{t,\ell}
  \;=\;
  \bigl\| \nabla_{\theta_{U\cap\ell}} L_t \bigr\|_2,
  \qquad
  \pi_{t,\ell}
  \;=\;
  \frac{g_{t,\ell}}{g_t+\varepsilon},
  \qquad
  \sum_{\ell}\pi_{t,\ell}=1.
\end{equation}

\begin{remark}
$\{\pi_{t,\ell}\}$ are used \emph{only after} $R_t^{\mathrm{grad}}=1$,
e.g.\ freeze / unfreeze ranking
$\ell^\star_t=\arg\max_{\ell}\pi_{t,\ell}$.
They do \textbf{not} enter~\eqref{eq:prank}--\eqref{eq:Rt}.
\end{remark}

%======================================================================
\section{Comparators and lead time}
\label{sec:lead}

Parallel monitors with the same OnlineRFPerm skeleton:
\begin{align}
  T_t^{\mathrm{MSE}}
  &=
  \mathrm{MSE}\bigl(f_{\mathrm{ref}}^{\mathrm{RF}},X_t,Y_t\bigr)
  - E_{\mathrm{ref}},
  \label{eq:mse}\\[0.25em]
  T_t^{\mathrm{MMD}}
  &=
  \widehat{\mathrm{MMD}}^2(X_{t-1},X_t)
  - e_{\mathrm{MMD}},
  \label{eq:mmd}\\[0.25em]
  T_t^{\mathrm{PO}}
  &=
  \mathrm{PO\text{-}gap}(t-1\!\to\! t)
  - e_{\mathrm{PO}}.
  \label{eq:po}
\end{align}
Let $t_{\bullet}=\min\{t\ge b: R_t^{\bullet}=1\}$ (else $\infty$).

\begin{definition}[Lead time]
\begin{equation}
\label{eq:lead}
  \Lead
  \;=\;
  t_{\mathrm{grad}} - t_{\mathrm{MSE}}.
\end{equation}
$\Lead<0$ means Grad rejects \emph{before} serving-MSE OnlineRFPerm.
\end{definition}

%======================================================================
\section{Algorithm}
\label{sec:algo}

\begin{enumerate}[leftmargin=*,itemsep=0.35em]
  \item Fit / pretrain $f_{\mathrm{ref}}$ on burn-in; freeze weights.
  \item For $t=0,\ldots,b-1$: compute $g_t$~\eqref{eq:gt}; update
        $e_{\mathrm{ref}}$~\eqref{eq:eref}; append $T_t$; no reject.
  \item For $t\ge b$:
  \begin{enumerate}[label=(\alph*),itemsep=0.2em]
    \item one backward on $L_t$; $g_t\leftarrow\|\gradU L_t\|_2$;
    \item $T_t\leftarrow g_t-e_{\mathrm{ref}}$; $p_t$ via~\eqref{eq:pewma};
    \item online FDR~\eqref{eq:alphat}--\eqref{eq:wealth}
          $\Rightarrow R_t^{\mathrm{grad}}$;
    \item if reject: record shares $\pi_{t,\ell}$~\eqref{eq:gl}
          for freeze-depth ranking;
    \item (parallel) update MSE / MMD / PO OnlineRFPerm.
  \end{enumerate}
\end{enumerate}

\subsection{Baseline alarm rate $= n_{\mathrm{alarm}}/n_{\mathrm{batch}}$}
\label{sec:null}

\begin{equation*}

\subsection{Freeze closed-loop}
\label{sec:freeze}

After the first Grad reject $t_{\mathrm{grad}}$, adapt a student with
one of:
\begin{itemize}[leftmargin=*,itemsep=0.2em]
  \item \texttt{always\_adapt}: all layers trainable;
  \item \texttt{freeze\_early}: freeze \texttt{fc1};
  \item \texttt{freeze\_low\_share}: freeze
        $\arg\min_{\ell}\pi_{t_{\mathrm{grad}},\ell}$;
  \item \texttt{no\_adapt}: freeze all after reject.
\end{itemize}
Report post-reject next-batch MSE and a FLOPs proxy
$(\#\text{trainable params})\times(\text{adapt steps})$.

\begin{table}[h]
\centering
\caption{Freeze closed-loop on synthetic (relative to \texttt{always\_adapt}).}
\label{tab:freeze-syn}
\begin{tabular}{lrr}
\toprule
policy & mse\_post (rel.) & FLOPs proxy (rel.) \\
\midrule
\texttt{always\_adapt}     & $0.9529$ ($1.00\times$) & $1.02\!\times\!10^6$ ($1.00\times$) \\
\texttt{freeze\_early}     & $1.6714$ ($1.75\times$) & $6.79\!\times\!10^5$ ($0.66\times$) \\
\texttt{freeze\_low\_share}& $1.0980$ ($1.15\times$) & $5.83\!\times\!10^5$ ($0.57\times$) \\
\texttt{no\_adapt}         & $13.56$  ($14.2\times$) & $\approx 0$ ($0.01\times$) \\
\bottomrule
\end{tabular}
\end{table}

\begin{table}[h]

($1.15\times$ MSE @ $0.57\times$ FLOPs). Full freeze collapses.

\section{Discussion}
\label{sec:disc}

\noindent\textbf{Conclusion (one sentence).}
It can detect the freezing point earlier than the other procedures
(MSE / MMD / PO OnlineRFPerm), with mean $\Lead\approx -3.3$ batches
across $35$ runs (MVP + adult / nomao).

\begin{enumerate}[leftmargin=*,itemsep=0.35em]
  \item \textbf{Engineering口径 is locked:} one unfrozen
        $\ell_2$ scalar + one OnlineRFPerm stream. No
        per-parameter / per-layer multiple testing.
  \item \textbf{MVP lead is real under that口径:}
        mean $\Lead\approx -3.2$, earlier $72\%$ of the time.
  \item \textbf{Sep18 replication:} adult / nomao add-on mean
        $\Lead\approx -3.6$ ($80\%$ earlier); pooled $35$ runs
        $\Lead\approx -3.34$ ($74\%$ earlier).
  \item \textbf{Baseline alarm rate:}
        $\mathrm{AR}=n_{\mathrm{alarm}}/n_{\mathrm{batch}}$
        (reference only --- \emph{not} Type-I FAR; updates preclude
        stationary-DGP claims). Grad $\approx$ MSE ($\sim 0.20$--$0.23$).
  \item \textbf{Grace} mildly delays first reject; higher AR is OK as baseline.
  \item \textbf{Freeze closed-loop} shows the intended product use:
        Grad reject $\to$ freeze by share / depth $\to$ MSE--FLOPs trade.
  \item \textbf{Next:} adapter / LoRA $\theta_U$ on larger models; couple with
        existing $\sqrt{\mathrm{PO}}$ adaptation gates.
\end{enumerate}
