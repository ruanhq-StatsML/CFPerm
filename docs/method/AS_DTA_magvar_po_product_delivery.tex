%======================================================================
\section{Literature review (PO-centred)}
%======================================================================

\paragraph{Potential outcomes and DR / CATE.}
The unit we monitor is not classifier confidence.
It sits in the potential-outcome / uplift stack:
outcome regressions $\hat\mu_w$, propensity $\hat\pi$, and a doubly robust
pseudo-outcome $\varphi$ whose regression $\hat\tau$ estimates CATE
\cite{rubin1974,robins1994,chernozhukov2018,kennedy2023,athey2019}.
PermuCATE-style RF nuisances instantiate that stack; unit PO-risk
$u=(\varphi-\hat\tau)^2$ is the natural residual risk for
``how unstable is the treatment effect fit on this row?''

\paragraph{Monitoring uplift / PO-risk under batch shift.}
Deployed uplift and targeting models need LIVE checks beyond holdout AUUC:
covariate shift on $X$, concept / outcome drift on $Y\mid X$, and
batch--outcome interaction when REF and LIVE are stacked with $W$
\cite{gama2014,quionero2009}.
FSDS-style monitors already couple domain signals on $X$ with
pseudo-outcome / PO-risk on stacked batches
\cite{monitorupliftfsds}.
Our MVP keeps that PO object and asks a narrower product question:
when mean risk rises, is the shift \emph{benign} or \emph{malignant}?

\paragraph{Why mean OOD / mean risk is not enough.}
Classical concept-drift and OOD detectors often collapse to a single
severity score (confidence, energy, MMD, domain AUC)
\cite{hendrycks2017,liu2020}.
Both cosmetic synonym drift and sparse integrity attacks can raise that
score --- so ``OOD$\uparrow\Rightarrow$retrain'' mixes two diseases and
two medicines (conformal vs freeze).
Magnitude alone inherits the same ambiguity.

\paragraph{Gap this MVP fills.}
We read \textbf{shape} of PermuCATE unit risk via Mag$\times$Var
($\mu=\mathbb{E}[u]$, $s=\mathrm{Std}(r_b)$ over bootstrap sub-envs):
uniform mass $\Rightarrow$ benign-like; heteroscedastic / mixture mass
$\Rightarrow$ malignant-like --- aligning with sparse-contamination
intuition in robust statistics and mixture shift.
On text boards with observed $Y$, we add \textbf{$Y$-flip on the attacked
subset} so integrity outcome break is explicit (same disease model as
tabular boards that break $Y$ on the hit set).
Product score $\mathrm{severity}=J+Y$-flip; actuators stay typed
(hold / conformal / VAI-freeze), optionally per 客群/region.

\noindent\textit{Positioning:}
not a new CATE estimator --- a \textbf{triage layer} on PermuCATE PO-risk
for benign vs malignant LIVE shift.

\section{How to separate benign vs malignant}
%======================================================================

\subsection{Two sensors}

\begin{enumerate}[leftmargin=1.3em,itemsep=0.4em]
  \item \textbf{Mag$\times$Var on $u$} --- shape of PermuCATE unit risk.
    Uniform drift: $\mu$ may rise, $s$ stays quiet $\Rightarrow$ benign-like.
    Mixture / sparse mass: $\mu$ and $s$ co-move $\Rightarrow$ malignant-like
    (hard on tabular; soft alone on text features).
  \item \textbf{$Y$-flip (text + response $Y$)} --- integrity break.
\end{enumerate}

\begin{table}[h]
\centering
\caption{$Y$-flip 口径 (text + PermuCATE).}
\label{tab:yflip}
\small
\begin{tabular}{@{}lll@{}}
\toprule
 & Text $X$ & Response $Y$ \\
\midrule
\textbf{Benign} & uniform synonym / back-translate & \textbf{unchanged} (flip $=0$) \\
\textbf{Malignant} & subset hard attack & \textbf{flip only on attacked rows} $Y\leftarrow 1-Y$ \\
\bottomrule
\end{tabular}
\end{table}

\begin{equation}
  \mathrm{Y\text{-}flip}
  =\frac{1}{n}\sum_i
  \mathbf{1}\{Y_i^{\mathrm{LIVE}}\neq Y_i^{\mathrm{REF}}\},
  \qquad
  \mathrm{severity}=J+\mathrm{Y\text{-}flip}.
\end{equation}
Sparse $\approx$ small flip; dense $\approx$ large flip.

\subsection{Decision rule (product)}
\begin{center}
\small
\begin{tabular}{@{}lll@{}}
\toprule
Read & Call & Actuator \\
\midrule
$Y$-flip $=0$, mild $J$ & benign / quiet & hold or conformal \\
mid $Y$-flip (+ Mag$\times$Var) & malignant sparse & VAI freeze (segment-local) \\
large $Y$-flip & malignant dense & broader freeze / retrain \\
\bottomrule
\end{tabular}
\end{center}

\noindent
客群/region: compute the same sensors inside finance / medicine / topic ---
local drug, not one global alarm.

%======================================================================
\section{Justify}
%======================================================================

\paragraph{Why this separates the right things.}
\begin{enumerate}[leftmargin=1.3em,itemsep=0.45em]
  \item \textbf{Objective $\Leftrightarrow$ disease definition.}
    Benign $=$ cosmetic $X$ drift with \emph{valid} $Y$.
    Malignant $=$ attack plus \emph{outcome break} on a subset.
    Keeping $Y$ for benign and flipping $Y$ only on attacked rows is that
    definition made measurable --- not an eval trick.
  \item \textbf{Same disease model as tabular PermuCATE.}
    Tabular malignant already breaks $Y$ on the attacked block while shifting
    $X$. Text $Y$-flip is the language-space twin, so one PO geometry
    ($u$, Mag$\times$Var) covers both.
  \item \textbf{Mag$\times$Var is necessary for shape, not always sufficient on text.}
    Synonym and attack both move features; pure $J$ can be soft
    (AUC($J$)$\approx 0.43$ pooled). Expected: without outcome break,
    $\varphi$-risk need not look malignant. Mag$\times$Var still flags
    heteroscedastic mass when it exists (tabular $J$: $-0.21$ vs $+114$).
    Severity adds $Y$-flip so integrity hits fire while benign stays conformal.
  \item \textbf{Wrong-drug cost is the product KPI.}
    Conformal on outcome break fails; freeze on synonym wastes recall.
    The two-sensor rule (shape $J$ + integrity $Y$-flip) is what prevents
    that swap --- which is the business point of separating benign vs malignant.
\end{enumerate}

\noindent\textit{One line:}
objective $=$ separate benign vs malignant;
method $=$ Mag$\times$Var on PermuCATE $u$ plus $Y$-flip;
justify $=$ cosmetic drift $\neq$ outcome break.
