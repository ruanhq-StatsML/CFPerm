\begin{abstract}
The unit risk for Mag$\times$Var is \textbf{not} $1-\hat p(Y\mid x)$.
It is your \textbf{PermuCATE} PO-risk: RF outcome models $\hat\mu_0,\hat\mu_1$,
propensity $\hat\pi$, DR pseudo-outcome $\varphi$, CATE $\hat\tau$, and
unit risk $u_i=(\varphi_i-\hat\tau_i)^2$ (batch risk $R=\mathbb{E}[u_i]$).
Magnitude $\mu=\mathbb{E}[u]$ and bootstrap variance $s=\mathrm{Std}(r_b)$
form a low-cost product triage proxy: even with bootstrap, the gate answers
hold / conformal / VAI-freeze before any heavy retrain.
\end{abstract}

\section{Canonical PO calculation (PermuCATE --- this is the logic)}

\paragraph{Treatment / outcome.}
Hop or batch indicator $W\in\{0,1\}$ (REF vs LIVE), reward / response $Y$, covariates $X$.

\paragraph{Nuisances (your stack; RF outcome by default).}
\begin{align}
  \hat\mu_w(x) &\approx \mathbb{E}[Y\mid X=x,W=w]
  &&\text{\texttt{RandomForestRegressor}},\\
  \hat\pi(x) &\approx \mathbb{P}(W=1\mid X=x)
  &&\text{RF / logistic classifier}.
\end{align}

\paragraph{Doubly robust pseudo-outcome (PermuCATE / \texttt{po\_risk}).}
\begin{equation}
\label{eq:phi}
  \varphi(Z)
  \;=\;
  \bigl(\hat\mu_1(X)-\hat\mu_0(X)\bigr)
  \;+\;
  \frac{\bigl(Y-\hat\mu_W(X)\bigr)\,(W-\hat\pi(X))}
       {\hat\pi(X)\,(1-\hat\pi(X))}\,.
\end{equation}

\paragraph{CATE and unit PO-risk.}
Fit $\hat\tau(X)$ by regressing $\varphi$ on $X$ (RF). Unit and batch risk:
\begin{equation}
\label{eq:unit-po}
  u_i \;=\; \bigl(\varphi_i-\hat\tau(X_i)\bigr)^2\,,
  \qquad
  R \;=\; \mathbb{E}[u_i]
  \;=\; \mathbb{E}\bigl[(\varphi-\hat\tau)^2\bigr]\,.
\end{equation}
This matches \texttt{permuCATE\_vimp.po\_risk} /
\texttt{permuCATE\_fit\_nuisance} (\texttt{model\_m=rf\_regressor}).

\paragraph{Code.}
\texttt{Python/src/as\_dta\_permucate\_magvar.py} $\rightarrow$
\texttt{permucate\_unit\_po} (cross-fit RF $\mu_0,\mu_1,\pi,\tau$).

\section{Mag$\times$Var on PermuCATE unit risk}

\paragraph{Magnitude / Variance.}
\begin{equation}
  \mu \;=\; \mathbb{E}[u_i]\,,
  \qquad
  r_b \;=\; \mathrm{mean}_{i\in b}(u_i)\ \text{(bootstrap window)},
  \qquad
  s \;=\; \mathrm{Std}(r_b)\,.
\end{equation}
Dashboard merge: $J=\Delta\mu\cdot(1+\kappa\Delta s_{+})$.

\paragraph{Why Mag$\times$Var (product justify).}
Uniform benign shift lifts $\{r_b\}$ together ($\mu\uparrow$, $s$ quiet).
Sparse malignant mass makes $\{r_b\}$ heteroscedastic ($\mu\uparrow$, $s\uparrow$).
Mag$\times$Var on \emph{PermuCATE} $u_i$ is the triage sensor---not a classifier accuracy proxy.
